# The FlatBuffers wire schema (SpellCircle.fbs) and the C++ header
# generated from it, shared by every decoder: the Qt-free SceneModel and
# the Qt NetworkManager verifier.
#
# The header is a BUILD ARTIFACT, generated into the build tree and never
# committed. flatc comes from the same vcpkg manifest as the runtime
# library, so generating costs no dependency a build does not already
# have, and the schema cannot drift from the header compiled against it.
#
# The Python modules under apps/python are a different case and stay
# committed: that package is installed and imported on its own — by
# TouchDesigner among others — with no CMake build in reach. They are
# regenerated by scripts/regen_flatbuffers.sh.

set(SCHEMA ${CMAKE_CURRENT_LIST_DIR}/SpellCircle.fbs)
set(SCHEMA_INCLUDE_DIR ${CMAKE_CURRENT_BINARY_DIR}/include)
set(SCHEMA_HEADER ${SCHEMA_INCLUDE_DIR}/SpellCircle_generated.h)

add_custom_command(
  OUTPUT ${SCHEMA_HEADER}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${SCHEMA_INCLUDE_DIR}
  COMMAND flatbuffers::flatc --cpp -o ${SCHEMA_INCLUDE_DIR} ${SCHEMA}
  DEPENDS ${SCHEMA}
  COMMENT "Generating SpellCircle_generated.h from the wire schema"
  VERBATIM)

# An add_custom_command() rule only enters the build graph through a
# target in its own directory, and an INTERFACE library builds nothing.
# Without this target the rule is orphaned and consumers report the
# header as missing with no rule to make it.
add_custom_target(SpellCircleSchemaGen DEPENDS ${SCHEMA_HEADER})

find_package(flatbuffers CONFIG REQUIRED)
add_library(SpellCircleSchema INTERFACE)
add_dependencies(SpellCircleSchema SpellCircleSchemaGen)

# Listing the generated header as an INTERFACE source is what makes a
# consumer wait for it. Neither an include directory nor an
# add_dependencies() on this library carries that ordering: an INTERFACE
# library builds nothing, so nothing propagates from it, and the first
# consumer to compile would race the generator.
target_sources(SpellCircleSchema INTERFACE ${SCHEMA_HEADER})

target_include_directories(SpellCircleSchema INTERFACE ${SCHEMA_INCLUDE_DIR})

target_link_libraries(SpellCircleSchema INTERFACE flatbuffers::flatbuffers)
